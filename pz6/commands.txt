Запуск: docker compose up -d
Вход:   docker compose exec db psql -U postgres
Выход из psql: \q
Остановка: docker compose down
Сброс БД с нуля: docker compose down -v && docker compose up -d


==================== ЗАДАНИЕ 1. ТАБЛИЦЫ И ДАННЫЕ ====================

CREATE TABLE students (
    id SERIAL PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    group_name VARCHAR(50) NOT NULL,
    email VARCHAR(100),
    scholarship NUMERIC(10,2)
);

CREATE TABLE grades (
    id SERIAL PRIMARY KEY,
    student_id INT NOT NULL REFERENCES students(id),
    teacher_username TEXT NOT NULL,
    subject VARCHAR(100) NOT NULL,
    grade INT NOT NULL CHECK (grade BETWEEN 0 AND 100)
);

==================== ЗАДАНИЕ 2. РОЛИ И СТОЛБЦЫ ====================

CREATE ROLE teacher_limited;
CREATE ROLE accountant_limited;

GRANT SELECT (id, full_name, group_name) ON students TO teacher_limited;
GRANT SELECT (id, scholarship) ON students TO accountant_limited;

-- проверка: преподаватель
SET ROLE teacher_limited;
SELECT id, full_name, group_name FROM students;      -- разрешено
SELECT scholarship FROM students;                     -- permission denied
UPDATE students SET scholarship = 0 WHERE id = 1;     -- permission denied
RESET ROLE;

-- проверка: бухгалтер
SET ROLE accountant_limited;
SELECT id, scholarship FROM students;                 -- разрешено
SELECT full_name FROM students;                       -- permission denied
RESET ROLE;


==================== ЗАДАНИЕ 3. VIEW ====================

CREATE VIEW student_public_info AS
SELECT id, full_name, group_name FROM students;

GRANT SELECT ON student_public_info TO teacher_limited;

-- отзыв прямого доступа к таблице (снимает и столбцовые права — выдаём их заново)
REVOKE SELECT ON students FROM teacher_limited;
GRANT SELECT (id, full_name, group_name) ON students TO teacher_limited;

-- проверка
SET ROLE teacher_limited;
SELECT * FROM student_public_info;                    -- 3 столбца, 4 строки
SELECT * FROM students;                               -- permission denied
RESET ROLE;


==================== ЗАДАНИЕ 4. RLS ====================

CREATE ROLE teacher01 LOGIN PASSWORD 'Teacher01_Strong!';
CREATE ROLE teacher02 LOGIN PASSWORD 'Teacher02_Strong!';

CREATE ROLE teacher_rls_role;
GRANT teacher_rls_role TO teacher01;
GRANT teacher_rls_role TO teacher02;

GRANT SELECT, UPDATE ON grades TO teacher_rls_role;

ALTER TABLE grades ENABLE ROW LEVEL SECURITY;

CREATE POLICY grades_select_policy ON grades
    FOR SELECT TO teacher_rls_role
    USING (teacher_username = current_user);

CREATE POLICY grades_update_policy ON grades
    FOR UPDATE TO teacher_rls_role
    USING (teacher_username = current_user)
    WITH CHECK (teacher_username = current_user);

\dp grades                                            -- в колонке Policies видны обе политики


==================== ЗАДАНИЕ 5. ПРОВЕРКИ ====================

-- Проверка 1: teacher01
\c postgres teacher01
SELECT * FROM grades;                                 -- 2 строки, id 1 и 2
UPDATE grades SET grade = 95 WHERE id = 1;            -- UPDATE 1
UPDATE grades SET grade = 100 WHERE id = 3;           -- UPDATE 0
UPDATE grades SET teacher_username = 'teacher02' WHERE id = 1;
                                                      -- ERROR: new row violates row-level security policy (WITH CHECK)

-- Проверка 2: teacher02
\c postgres teacher02
SELECT * FROM grades;                                 -- 2 строки, id 3 и 4, оценка 3 осталась 92

-- обратно в админа
