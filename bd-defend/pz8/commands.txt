Запуск: docker compose up -d
Вход:   docker compose exec db psql -U postgres -d university
(setup.sql при первом старте уже создал обе таблицы и подключил pgcrypto — этапы 1-3.
 Чтобы прогнать руками: docker compose down -v, закомментировать setup.sql в compose, up -d)


==================== ЭТАП 1. ОТКРЫТОЕ ХРАНЕНИЕ ====================

CREATE TABLE students_security (
    id SERIAL PRIMARY KEY,
    student_name TEXT NOT NULL,
    document_number TEXT NOT NULL
);
INSERT INTO students_security (student_name, document_number) VALUES
    ('Иван Петров', 'DOC-1001'),
    ('Анна Сидорова', 'DOC-1002'),
    ('Алексей Смирнов', 'DOC-1003');
SELECT * FROM students_security;                      -- 3 строки, номера читаются


==================== ЭТАП 2. PGCRYPTO ====================

CREATE EXTENSION IF NOT EXISTS pgcrypto;
SELECT extname, extversion FROM pg_extension WHERE extname = 'pgcrypto';   -- pgcrypto 1.3
SELECT encode(pgp_sym_encrypt('TEST-123', 'class-demo-key'), 'hex') AS encrypted_value;
-- повтори ещё раз: hex будет другой, потому что внутри случайная соль


==================== ЭТАП 3. ЗАШИФРОВАННАЯ ТАБЛИЦА ====================

CREATE TABLE students_encrypted (
    id SERIAL PRIMARY KEY,
    student_name TEXT NOT NULL,
    encrypted_document BYTEA NOT NULL
);
INSERT INTO students_encrypted (student_name, encrypted_document) VALUES
    ('Иван Петров',     pgp_sym_encrypt('DOC-1001', 'class-demo-key')),
    ('Анна Сидорова',   pgp_sym_encrypt('DOC-1002', 'class-demo-key')),
    ('Алексей Смирнов', pgp_sym_encrypt('DOC-1003', 'class-demo-key'));

SELECT id, student_name, encode(encrypted_document, 'hex') AS encrypted_value
FROM students_encrypted;                              -- hex-мусор по 74 байта вместо DOC-100x


==================== ЭТАП 4. РАСШИФРОВАНИЕ ====================

SELECT id, student_name, pgp_sym_decrypt(encrypted_document, 'class-demo-key') AS document_number
FROM students_encrypted;                              -- DOC-1001, DOC-1002, DOC-1003

SELECT pgp_sym_decrypt(encrypted_document, 'incorrect-demo-key')
FROM students_encrypted;                              -- ERROR: Wrong key or corrupt data


==================== ЭТАП 5. СРАВНЕНИЕ ====================

SELECT student_name, document_number FROM students_security;
SELECT student_name, encode(encrypted_document, 'hex') AS encrypted_document FROM students_encrypted;

-- поиск по значению
SELECT * FROM students_security WHERE document_number = 'DOC-1002';       -- индекс, быстро
SELECT id, student_name FROM students_encrypted
WHERE pgp_sym_decrypt(encrypted_document, 'class-demo-key') = 'DOC-1002'; -- расшифровка каждой строки
SELECT count(*) FROM students_encrypted
WHERE encrypted_document = pgp_sym_encrypt('DOC-1002', 'class-demo-key'); -- 0, шифротекст не совпадает


docker compose down
