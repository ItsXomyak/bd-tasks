==================== ТАБЛИЦЫ ====================

CREATE TABLE students (id SERIAL PRIMARY KEY, full_name VARCHAR(100), group_name VARCHAR(50), email VARCHAR(100));
CREATE TABLE teachers (id SERIAL PRIMARY KEY, full_name VARCHAR(100), department VARCHAR(100));
CREATE TABLE groups   (id SERIAL PRIMARY KEY, group_name VARCHAR(50), specialty VARCHAR(100));
CREATE TABLE grades   (id SERIAL PRIMARY KEY, student_id INT, teacher_id INT, subject VARCHAR(100), grade INT);
CREATE TABLE payments (id SERIAL PRIMARY KEY, student_id INT, amount DECIMAL(10,2), payment_date DATE);
CREATE TABLE reports  (id SERIAL PRIMARY KEY, report_name VARCHAR(100), created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP);
CREATE TABLE users    (id SERIAL PRIMARY KEY, username VARCHAR(100), role_name VARCHAR(50));

==================== ЗАДАНИЕ 3. ПРАВА РОЛЯМ ====================

-- 3.1 студент
GRANT SELECT ON students TO student_role;
GRANT SELECT ON grades TO student_role;

-- 3.2 преподаватель
GRANT SELECT ON students TO teacher_role;
GRANT SELECT ON teachers TO teacher_role;
GRANT SELECT ON groups TO teacher_role;
GRANT SELECT, INSERT, UPDATE ON grades TO teacher_role;

-- 3.3 бухгалтер
GRANT SELECT, INSERT, UPDATE ON payments TO accountant_role;

-- 3.4 деканат
GRANT SELECT ON students TO dean_role;
GRANT SELECT ON teachers TO dean_role;
GRANT SELECT ON groups TO dean_role;
GRANT SELECT, INSERT, UPDATE ON grades TO dean_role;
GRANT SELECT, INSERT, UPDATE ON reports TO dean_role;

-- 3.5 аналитик
GRANT SELECT ON students TO analyst_role;
GRANT SELECT ON grades TO analyst_role;
GRANT SELECT ON payments TO analyst_role;
GRANT SELECT ON reports TO analyst_role;

-- INSERT в таблицу с SERIAL требует права на последовательность
GRANT USAGE ON SEQUENCE grades_id_seq TO teacher_role, dean_role;
GRANT USAGE ON SEQUENCE payments_id_seq TO accountant_role;
GRANT USAGE ON SEQUENCE reports_id_seq TO dean_role;

-- админ
GRANT ALL ON ALL TABLES IN SCHEMA public TO admin_role;
GRANT ALL ON ALL SEQUENCES IN SCHEMA public TO admin_role;

==================== ЗАДАНИЕ 5. ИЕРАРХИЯ ====================

CREATE ROLE employee_role;
GRANT SELECT ON students TO employee_role;

CREATE ROLE senior_employee_role;
GRANT employee_role TO senior_employee_role;
GRANT SELECT ON teachers TO senior_employee_role;
GRANT SELECT ON groups TO senior_employee_role;

-- проверка наследования
SET ROLE senior_employee_role;
SELECT * FROM students;                               -- разрешено (унаследовано от employee_role)
SELECT * FROM teachers;                               -- разрешено (своё)
SELECT * FROM payments;                               -- permission denied
RESET ROLE;
SET ROLE employee_role;
SELECT * FROM students;                               -- разрешено
SELECT * FROM teachers;                               -- permission denied (базовая роль не видит)
RESET ROLE;

==================== ИТОГОВАЯ МАТРИЦА ИЗ БД ====================

\c university postgres
SELECT grantee, table_name, string_agg(privilege_type, ',' ORDER BY privilege_type) AS privs
FROM information_schema.role_table_grants
WHERE grantee LIKE '%_role' AND grantee <> 'admin_role'
GROUP BY grantee, table_name ORDER BY grantee, table_name;
